# Platform-specific thread subsystem deps that may not exist on all OS targets.
# These are collected into variables so entrypoints can depend on them
# conditionally without requiring if(TARGET) guards at every call site.
set(_pthread_os_cancel_support_dep)
if(TARGET libc.src.__support.threads.${LIBC_TARGET_OS}.cancel_support)
  set(_pthread_os_cancel_support_dep
    libc.src.__support.threads.${LIBC_TARGET_OS}.cancel_support)
endif()

set(_pthread_os_sched_support_dep)
if(TARGET libc.src.__support.threads.${LIBC_TARGET_OS}.sched_support)
  set(_pthread_os_sched_support_dep
    libc.src.__support.threads.${LIBC_TARGET_OS}.sched_support)
endif()

set(_pthread_os_thread_lifecycle_dep)
if(TARGET libc.src.__support.threads.${LIBC_TARGET_OS}.thread_lifecycle)
  set(_pthread_os_thread_lifecycle_dep
    libc.src.__support.threads.${LIBC_TARGET_OS}.thread_lifecycle)
endif()

set(_pthread_os_thread_registry_dep)
if(TARGET libc.src.__support.threads.${LIBC_TARGET_OS}.thread_registry)
  set(_pthread_os_thread_registry_dep
    libc.src.__support.threads.${LIBC_TARGET_OS}.thread_registry)
endif()

set(_pthread_os_cpuclock_support_dep)
if(TARGET libc.src.__support.threads.${LIBC_TARGET_OS}.cpuclock_support)
  set(_pthread_os_cpuclock_support_dep
    libc.src.__support.threads.${LIBC_TARGET_OS}.cpuclock_support)
endif()

set(_pthread_os_rwlock_dep)
if(TARGET libc.src.__support.threads.${LIBC_TARGET_OS}.rwlock)

    libc.src.__support.threads.${LIBC_TARGET_OS}.rwlock)
